cmake_minimum_required(VERSION 3.20)
project(BlackjackEngine LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(blackjack_engine INTERFACE)
target_include_directories(blackjack_engine INTERFACE ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(blackjack_tests tests.cpp)
target_link_libraries(blackjack_tests PRIVATE blackjack_engine)
enable_testing()
add_test(NAME blackjack_rules COMMAND blackjack_tests)
